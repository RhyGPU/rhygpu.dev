3.2 â€” Memory Scoring Algorithm (Executed by Rust Engine, Not LLM)
rust
struct MemoryScorer {
    embedding_model: MiniLMEmbedder,  // ONNX, runs locally
    existing_memories: Vec<Memory>,   // all recent + core entries
    character_needs: MaslowLevels,    // current need satisfaction levels
}

impl MemoryScorer {
    fn score(&self, new_memory: &Memory) -> f32 {
        // 1. Emotional intensity (from the tag category)
        let emotional_score = match new_memory.tag.as_str() {
            "identity_violation" | "betrayal" | "near_death" => 0.95,
            "trauma_trigger" | "control_gain" | "trust_break" => 0.85,
            "bonding" | "trust_building" | "intimacy" | "compassion" => 0.75,
            "introduction" | "dynamic_establishment" | "orientation" => 0.60,
            "boundary_setting" | "conflict_minor" => 0.50,
            "routine" | "small_talk" | "observation" => 0.30,
            _ => 0.50,
        };

        // 2. Novelty (cosine distance from average of existing memories)
        let new_embedding = self.embedding_model.embed(&new_memory.content);
        let avg_existing = self.average_embedding();
        let novelty_score = 1.0 - cosine_similarity(&new_embedding, &avg_existing);
        // Clamp: if no existing memories, novelty = 1.0

        // 3. Goal relevance (match against character's current needs)
        let goal_score = self.compute_goal_relevance(&new_memory.tag);

        // 4. Repetition discount
        let similar_count = self.count_similar_memories(&new_memory);
        let repetition_discount = 1.0 / (1.0 + similar_count as f32 * 0.3);

        // Weighted composite
        let raw_score = (emotional_score * 0.4) + (novelty_score * 0.3) + (goal_score * 0.3);
        let final_score = (raw_score * repetition_discount).clamp(0.0, 1.0);

        final_score
    }

    fn compute_goal_relevance(&self, tag: &str) -> f32 {
        // If belonging is starved (low), bonding events score higher
        // If safety is threatened, threat-related events score higher
        let needs = &self.character_needs;
        match tag {
            "bonding" | "trust_building" | "intimacy" | "compassion" => {
                let belonging_deficit = 1.0 - (needs.belong as f32 / 100.0);
                belonging_deficit.clamp(0.3, 1.0)
            },
            "threat" | "danger" | "fear" => {
                let safety_deficit = 1.0 - (needs.safety as f32 / 100.0);
                safety_deficit.clamp(0.3, 1.0)
            },
            _ => 0.5, // neutral
        }
    }
}

    if recent_memories.is_empty() {
        return 1.0; // first memory is always novel
    }

    let tag = &new_memory.tag;
    let content_words: HashSet<&str> = new_memory.content.split_whitespace().collect();

    // 1. Tag novelty: how rare is this tag in recent memory?
    let same_tag_count = recent_memories.iter()
        .filter(|m| m.tag == *tag)
        .count();
    let tag_novelty = 1.0 - (same_tag_count as f32 / recent_memories.len() as f32);

    // 2. Content novelty: Jaccard distance from the most similar recent memory
    let mut max_jaccard = 0.0_f32;
    for mem in recent_memories.iter() {
        let mem_words: HashSet<&str> = mem.content.split_whitespace().collect();
        let intersection = content_words.intersection(&mem_words).count();
        let union = content_words.union(&mem_words).count();
        let jaccard = intersection as f32 / union as f32;
        if jaccard > max_jaccard {
            max_jaccard = jaccard;
        }
    }
    let content_novelty = 1.0 - max_jaccard; // high similarity â†’ low novelty

    // 3. Combine (tag is a strong signal, content refines)
    let raw = (tag_novelty * 0.6) + (content_novelty * 0.4);
    raw.clamp(0.0, 1.0)
}
